=== -- [Q09] 매장별 평균 메뉴 가격 및 등록된 메뉴 수 집계 (AVG, COUNT, GROUP BY) ===
[SQL]:
SELECT s.name AS store_name, COUNT(m.id) AS menu_count, AVG(m.price) AS avg_menu_price
FROM store s
INNER JOIN menu m ON s.id = m.store_id
GROUP BY s.id, s.name;
