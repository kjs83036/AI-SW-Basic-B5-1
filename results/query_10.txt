=== -- [Q10] 매장별 결제 완료(PAID) 누적 매출액 집계 (SUM, GROUP BY, ORDER BY) ===
[SQL]:
SELECT s.name AS store_name, SUM(o.total_amount) AS total_sales
FROM store s
INNER JOIN orders o ON s.id = o.store_id
WHERE o.status = 'PAID'
GROUP BY s.id, s.name
ORDER BY total_sales DESC;
