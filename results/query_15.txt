=== -- [Q15] 취소된(CANCELLED) 주문의 상세 주문 항목 기록 삭제 (DELETE) ===
[SQL]:
DELETE FROM order_item
WHERE order_id IN (SELECT id FROM orders WHERE status = 'CANCELLED');
